// === Monitor de humedad del suelo - Arduino UNO R3 ===

const int PIN_SENSOR = A0;        // Salida analógica del sensor
const int PIN_LED    = 13;        // LED integrado del UNO R3 (o uno externo)

// Calibración (valores de ejemplo, ver nota abajo)
const int VALOR_SECO   = 520;     // Lectura con el sensor al aire / tierra seca
const int VALOR_MOJADO = 260;     // Lectura con el sensor en agua / tierra empapada

const int UMBRAL_PORCENTAJE = 40; // Por debajo de este % se considera seca

void setup() {
  Serial.begin(9600);
  pinMode(PIN_LED, OUTPUT);

  Serial.println("=== Monitor de humedad del suelo - UNO R3 ===");
}

void loop() {
  int lectura = analogRead(PIN_SENSOR);

  int porcentaje = map(lectura, VALOR_SECO, VALOR_MOJADO, 0, 100);
  porcentaje = constrain(porcentaje, 0, 100);

  Serial.print("Lectura: ");
  Serial.print(lectura);
  Serial.print("  |  Humedad: ");
  Serial.print(porcentaje);
  Serial.print("%  |  Estado: ");

  if (porcentaje < UMBRAL_PORCENTAJE) {
    Serial.println("TIERRA SECA -> Se recomienda regar");
    digitalWrite(PIN_LED, HIGH);   // Alerta encendida
  } else {
    Serial.println("TIERRA HUMEDA -> No necesita riego");
    digitalWrite(PIN_LED, LOW);    // Alerta apagada
  }

  delay(1000);  // Una lectura por segundo
}
